N1 := 20;
N2 := 10;
N3 := 14;
SLK := 1.0;
TRK := 2.0;
BER := 0.5;
N4 := 120;
NDD := 140;
MAHD := EMA(CLOSE, NDD);
DIRP := IF(REF(CLOSE, 1) > REF(MAHD, 1), 1, 0 - 1);
OKL := DIRP >= 0;
OKS := DIRP <= 0;
MAL := EMA(CLOSE, N1);
TRU := MAX(MAX(HIGH - LOW, ABS(REF(CLOSE, 1) - HIGH)), ABS(REF(CLOSE, 1) - LOW));
ATR14 := SMA(TRU, N3, 1);
XUP := CLOSE > MAL AND REF(CLOSE, 1) < REF(MAL, 1);
XDN := CLOSE < MAL AND REF(CLOSE, 1) > REF(MAL, 1);
STRB := CLOSE > OPEN AND CLOSE >= (HIGH + LOW) / 2;
STRR := CLOSE < OPEN AND CLOSE <= (HIGH + LOW) / 2;
AL0 := OKL AND CLOSE > MAL AND LOW <= MAL AND STRB AND XUP;
AS0 := OKS AND CLOSE < MAL AND HIGH >= MAL AND STRR AND XDN;
HHN := REF(HHV(HIGH, N2), 1);
LLN := REF(LLV(LOW, N2), 1);
BL0 := OKL AND CLOSE > HHN AND CLOSE > OPEN;
BS0 := OKS AND CLOSE < LLN AND CLOSE < OPEN;
BK0 := AL0 OR BL0;
SK0 := AS0 OR BS0;
NB := BARSLAST(BK0);
NS := BARSLAST(SK0);
EPB := REF(CLOSE, NB);
EPS := REF(CLOSE, NS);
EAB := REF(ATR14, NB);
EAS := REF(ATR14, NS);
HIB := HHV(CLOSE, NB + 1);
LOB := LLV(CLOSE, NS + 1);
COREL := MAX(EPB - SLK * EAB, HIB - TRK * EAB);
BEL := HIB >= EPB + BER * EAB;
STOPL := IF(BEL, MAX(COREL, EPB), COREL);
CORES := MIN(EPS + SLK * EAS, LOB + TRK * EAS);
BES := LOB <= EPS - BER * EAS;
STOPS := IF(BES, MIN(CORES, EPS), CORES);
XLONG := NB >= 1 AND CLOSE <= STOPL;
XSHORT := NS >= 1 AND CLOSE >= STOPS;
LOSSL := XLONG AND CLOSE < EPB;
WINL := XLONG AND CLOSE >= EPB;
LOSSS := XSHORT AND CLOSE > EPS;
WINS := XSHORT AND CLOSE <= EPS;
LOSX := LOSSL OR LOSSS;
WINX := WINL OR WINS;
BGN := BARSCOUNT(CLOSE) < 2;
LSCNT := COUNT(LOSX, BARSLAST(WINX OR BGN) + 1);
B3 := BARSLAST(LSCNT >= 3 AND REF(LSCNT, 1) < 3);
COOLOK := B3 >= N4;
ENTERLONG: BK0 AND COOLOK;
EXITLONG: XLONG;
ENTERSHORT: SK0 AND COOLOK;
EXITSHORT: XSHORT;
