FSDIR := IF(CLOSE > EMA(CLOSE, 140), 1, 0 - 1);
FSHTF := REF(FSDIR, 1);
FSE20 := EMA(CLOSE, 20);
FSUP := CLOSE > FSE20 AND REF(CLOSE, 1) < REF(FSE20, 1);
FSDN := CLOSE < FSE20 AND REF(CLOSE, 1) > REF(FSE20, 1);
FSTL := LOW <= FSE20;
FSTS := HIGH >= FSE20;
FSBL := CLOSE > OPEN AND CLOSE >= (HIGH + LOW) / 2;
FSBR := CLOSE < OPEN AND CLOSE <= (HIGH + LOW) / 2;
FSKL := CLOSE > REF(HHV(HIGH, 10), 1) AND CLOSE > OPEN;
FSKS := CLOSE < REF(LLV(LOW, 10), 1) AND CLOSE < OPEN;
FSKB := FSKL AND REF(FSKL, 1) = 0;
FSSK := FSKS AND REF(FSKS, 1) = 0;
FSLON := FSHTF >= 0 AND ((FSUP AND FSTL AND FSBL) OR FSKB);
FSSHT := FSHTF <= 0 AND ((FSDN AND FSTS AND FSBR) OR FSSK);
DRAWICON(FSLON, LOW * 0.998, 1);
DRAWICON(FSSHT, HIGH * 1.002, 2);
